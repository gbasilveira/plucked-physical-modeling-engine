cmake_minimum_required(VERSION 3.25)
project(piano_mvp_v4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(CORE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(piano_core STATIC
  ${CORE}/engine/src/osc/oscillator.cpp
  ${CORE}/engine/src/osc/mode_bank.cpp
  ${CORE}/engine/src/string/string_model.cpp
  ${CORE}/engine/src/bridge/model.cpp
  ${CORE}/engine/src/bridge/terms.cpp
  ${CORE}/engine/src/nonlinear/sav.cpp
  ${CORE}/engine/src/nonlinear/tension.cpp
  ${CORE}/engine/src/exciters/hammer.cpp
  piano.cpp tension_note.cpp bridge_model.cpp string_loss.cpp grand_board.cpp ${CMAKE_CURRENT_SOURCE_DIR}/../piano-v3/soundboard.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/../piano-v2/preset.cpp
  ${CORE}/engine/src/simd/denormals.cpp)
target_include_directories(piano_core PUBLIC ${CORE}/engine/include ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(piano_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../piano-v2)
target_include_directories(piano_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../piano-v3)
foreach(module osc string bridge nonlinear exciters simd)
  target_include_directories(piano_core PUBLIC ${CORE}/engine/src/${module}/include)
endforeach()
add_executable(piano-render render.cpp)
target_link_libraries(piano-render PRIVATE piano_core)
add_executable(piano-midi-render midi_render.cpp)
target_link_libraries(piano-midi-render PRIVATE piano_core)
add_executable(piano-sympathetic-demo sympathetic_demo.cpp)
target_link_libraries(piano-sympathetic-demo PRIVATE piano_core)
add_executable(piano-unpedalled-demo unpedalled_demo.cpp)
target_link_libraries(piano-unpedalled-demo PRIVATE piano_core)
add_executable(piano-hammer-audit hammer_audit.cpp)
target_link_libraries(piano-hammer-audit PRIVATE piano_core)
enable_testing()
add_test(NAME piano-hammer-contact COMMAND piano-hammer-audit)
set_tests_properties(piano-hammer-contact PROPERTIES TIMEOUT 30)
add_executable(piano-check check.cpp)
target_link_libraries(piano-check PRIVATE piano_core)
add_test(NAME piano-behaviour COMMAND piano-check)
set_tests_properties(piano-behaviour PROPERTIES TIMEOUT 400)
add_executable(bridge-check bridge_check.cpp)
target_link_libraries(bridge-check PRIVATE piano_core)
add_test(NAME bridge-energy COMMAND bridge-check)
set_tests_properties(bridge-energy PROPERTIES TIMEOUT 120)
add_executable(piano-tension-check tension_check.cpp)
target_link_libraries(piano-tension-check PRIVATE piano_core)
add_test(NAME piano-tension-energy COMMAND piano-tension-check)
set_tests_properties(piano-tension-energy PROPERTIES TIMEOUT 120)
add_executable(piano-tension-measure tension_measure.cpp)
target_link_libraries(piano-tension-measure PRIVATE piano_core)
add_executable(piano-tension-bench tension_bench.cpp)
target_link_libraries(piano-tension-bench PRIVATE piano_core)
set(CLAP_INCLUDE_DIR "" CACHE PATH "Directory containing clap/clap.h (optional plug-in)")
if(CLAP_INCLUDE_DIR)
  add_library(piano-mvp-v4 MODULE clap.cpp)
  target_include_directories(piano-mvp-v4 PRIVATE ${CLAP_INCLUDE_DIR})
  target_link_libraries(piano-mvp-v4 PRIVATE piano_core)
  set_target_properties(piano-mvp-v4 PROPERTIES PREFIX "" SUFFIX ".clap")
  if(UNIX AND NOT APPLE)
    add_executable(piano-host-check host_check.cpp)
    target_include_directories(piano-host-check PRIVATE ${CLAP_INCLUDE_DIR})
    target_link_libraries(piano-host-check PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME piano-clap-playback COMMAND piano-host-check $<TARGET_FILE:piano-mvp-v4>)
    set_tests_properties(piano-clap-playback PROPERTIES TIMEOUT 120)
  endif()
endif()
option(PME_MVP_BUILD_LV2 "Build the Linux LV2 bundle" ON)
if(PME_MVP_BUILD_LV2 AND UNIX AND NOT APPLE)
  find_path(LV2_INCLUDE_DIR lv2/core/lv2.h REQUIRED)
  set(PIANO_LV2_BUNDLE "${CMAKE_CURRENT_BINARY_DIR}/piano-mvp-v4.lv2")
  file(MAKE_DIRECTORY "${PIANO_LV2_BUNDLE}")
  configure_file(lv2/manifest.ttl "${PIANO_LV2_BUNDLE}/manifest.ttl" COPYONLY)
  configure_file(lv2/piano.ttl "${PIANO_LV2_BUNDLE}/piano.ttl" COPYONLY)
  configure_file(lv2/README.txt "${PIANO_LV2_BUNDLE}/README.txt" COPYONLY)
  configure_file("${CORE}/LICENSE" "${PIANO_LV2_BUNDLE}/LICENSE.txt" COPYONLY)
  add_library(piano-mvp-v4-lv2 MODULE lv2.cpp)
  target_include_directories(piano-mvp-v4-lv2 PRIVATE ${LV2_INCLUDE_DIR})
  target_link_libraries(piano-mvp-v4-lv2 PRIVATE piano_core)
  set_target_properties(piano-mvp-v4-lv2 PROPERTIES PREFIX "" OUTPUT_NAME "piano-mvp-v4"
    LIBRARY_OUTPUT_DIRECTORY "${PIANO_LV2_BUNDLE}")
  add_custom_target(piano-lv2-package
    COMMAND "${CMAKE_COMMAND}" -E tar cf
      "piano-mvp-v4-lv2-linux-${CMAKE_SYSTEM_PROCESSOR}.zip" --format=zip -- piano-mvp-v4.lv2
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    DEPENDS piano-mvp-v4-lv2)
  add_executable(piano-lv2-check lv2_host_check.cpp)
  target_include_directories(piano-lv2-check PRIVATE ${LV2_INCLUDE_DIR})
  target_link_libraries(piano-lv2-check PRIVATE piano_core ${CMAKE_DL_LIBS})
  set_target_properties(piano-lv2-check PROPERTIES ENABLE_EXPORTS ON)
  add_test(NAME piano-lv2-playback COMMAND piano-lv2-check $<TARGET_FILE:piano-mvp-v4-lv2>)
  set_tests_properties(piano-lv2-playback PROPERTIES TIMEOUT 120)
  find_program(LV2_VALIDATE lv2_validate)
  if(LV2_VALIDATE)
    add_test(NAME piano-lv2-metadata COMMAND ${LV2_VALIDATE}
      "${PIANO_LV2_BUNDLE}/manifest.ttl" "${PIANO_LV2_BUNDLE}/piano.ttl")
  endif()
endif()
